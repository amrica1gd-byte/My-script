local function spawnFlag()
    local character = player.Character or player.CharacterAdded:Wait()
    local root = character:WaitForChild("HumanoidRootPart")
    if spawnedFlag then spawnedFlag:Destroy() end
    spawnedFlag = Instance.new("Model")
    spawnedFlag.Name = "ClientFlag_FromMapExport"

    -- Place the original assembly 12 studs in front of the character, with its base near ground level.
    local look = root.CFrame.LookVector
    local yaw = math.atan2(-look.X, -look.Z)
    local groundY = math.max(0, root.Position.Y - 3)
    local origin = Vector3.new(root.Position.X + look.X * 12, groundY, root.Position.Z + look.Z * 12)
    local targetPivot = CFrame.new(origin) * CFrame.Angles(0, yaw, 0)

    for _, d in ipairs(DATA) do
        local part = Instance.new(d.class == "WedgePart" and "WedgePart" or "Part")
        part.Name = d.name
        part.Size = d.size
        part.Color = d.color
        part.Material = Enum.Material[d.material] or Enum.Material.Plastic
        if part:IsA("Part") then
            part.Shape = Enum.PartType[d.shape] or Enum.PartType.Block
        end
        part.Transparency = d.transparency
        part.Reflectance = d.reflectance
        part.CastShadow = d.castShadow
        part.Anchored = true
        part.CanCollide = false
        part.CanTouch = false
        part.CanQuery = false
        local c = d.cf
        local localCF = CFrame.new(c[1], c[2], c[3], c[4], c[5], c[6], c[7], c[8], c[9], c[10], c[11], c[12])
        part.CFrame = targetPivot * localCF
        part.Parent = spawnedFlag

        if d.mesh then
            local mesh = Instance.new("SpecialMesh")
            mesh.MeshType = Enum.MeshType[d.mesh.meshType] or Enum.MeshType.FileMesh
            if d.mesh.meshId ~= "" then mesh.MeshId = d.mesh.meshId end
            if d.mesh.textureId ~= "" then mesh.TextureId = d.mesh.textureId end
            mesh.Scale = d.mesh.scale
            mesh.Offset = d.mesh.offset
            mesh.Parent = part
        end

        if d.light then
            local light = Instance.new("PointLight")
            light.Enabled = d.light.enabled
            light.Shadows = d.light.shadows
            light.Brightness = d.light.brightness
            light.Range = d.light.range
            light.Color = d.light.color
            light.Parent = part
        end
    end
    spawnedFlag.Parent = workspace
end

spawnFlag()
